cmake_minimum_required(VERSION 3.10)
project(remote_phy VERSION 0.1 LANGUAGES C)
add_library(remote_phy STATIC phy.c select.c)
target_include_directories(remote_phy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(remote_phy PROPERTIES C_STANDARD 99)
target_compile_options(remote_phy PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(remote_phy PRIVATE drivers/posix/fd.c drivers/posix/tcp.c)
    find_package(Threads REQUIRED)
    add_library(remote_phy_mux STATIC mux/queue.c mux/life.c mux/channel.c mux/pump.c mux/wait.c)
    target_link_libraries(remote_phy_mux PUBLIC remote_phy Threads::Threads)
    set_target_properties(remote_phy_mux PROPERTIES C_STANDARD 99)
    target_compile_options(remote_phy_mux PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    enable_testing()
    if(TARGET remote_phy_mux)
        add_executable(remote-phy-mux tests/mux.c)
        target_link_libraries(remote-phy-mux PRIVATE remote_phy_mux)
        add_test(NAME remote-phy-mux COMMAND remote-phy-mux)
        set_tests_properties(remote-phy-mux PROPERTIES TIMEOUT 20)
        add_executable(remote-phy-mux-failure tests/mux-failure.c)
        target_link_libraries(remote-phy-mux-failure PRIVATE remote_phy_mux)
        add_test(NAME remote-phy-mux-failure COMMAND remote-phy-mux-failure)
        set_tests_properties(remote-phy-mux-failure PROPERTIES TIMEOUT 10)
    endif()
    add_executable(remote-phy-core tests/core.c)
    target_link_libraries(remote-phy-core PRIVATE remote_phy)
    add_test(NAME remote-phy-core COMMAND remote-phy-core)
endif()
