cmake_minimum_required(VERSION 3.10)
project(fvm_owned_operations LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(fvm_owned owned.c)
target_include_directories(fvm_owned PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(fvm_owned PUBLIC Threads::Threads)
target_compile_options(fvm_owned PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(fvm_owned_table table_owner.c ../table/table.c)
target_include_directories(fvm_owned_table PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../table")
target_link_libraries(fvm_owned_table PUBLIC fvm_owned)
target_compile_options(fvm_owned_table PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(owned_splice_example example_splice.c)
target_link_libraries(owned_splice_example PRIVATE fvm_owned_table)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME owned_splice_tutorial COMMAND owned_splice_example)
    add_executable(owned_test test_owned.c)
    target_link_libraries(owned_test PRIVATE fvm_owned)
    add_test(NAME owned_operation_contention COMMAND owned_test)
    set_tests_properties(owned_operation_contention PROPERTIES TIMEOUT 20)
    add_executable(owned_table_test test_table_owner.c)
    target_link_libraries(owned_table_test PRIVATE fvm_owned_table)
    add_test(NAME owned_table_transactions COMMAND owned_table_test)
    set_tests_properties(owned_table_transactions PROPERTIES TIMEOUT 20)
endif()
