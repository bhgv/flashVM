cmake_minimum_required(VERSION 3.10)
project(fvm_hybrid_table VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(fvm_hybrid_table table.c)
target_include_directories(fvm_hybrid_table PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fvm_hybrid_table PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(fvm_table_test test_table.c)
  target_link_libraries(fvm_table_test PRIVATE fvm_hybrid_table)
  add_test(NAME fvm_table_semantics COMMAND fvm_table_test)
endif()
