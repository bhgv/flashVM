cmake_minimum_required(VERSION 3.10)
project(inferno_native_libraries VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS ON)
find_package(Threads REQUIRED)
find_package(PythonInterp 3 REQUIRED)
set(INFERNO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src" CACHE PATH "Exact Inferno source root")
set(INFERNO_HOST_SUPPORT "" CACHE PATH "Explicit shared bootstrap host support source root")
if(NOT EXISTS "${INFERNO_HOST_SUPPORT}/prepare_fmt.py")
    message(FATAL_ERROR "Set INFERNO_HOST_SUPPORT to the tools/limbo-bootstrap directory")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Initial native library acceptance requires Linux64")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(inferno_arch amd64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(inferno_arch arm64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(riscv64|rv64)$")
    set(inferno_arch riscv64)
else()
    message(FATAL_ERROR "Unsupported native64 architecture ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(inferno_headers "${INFERNO_SOURCE}/Linux/${inferno_arch}/include" "${INFERNO_SOURCE}/include")
add_subdirectory(lib9)
add_subdirectory(libstyx)
enable_testing()
add_executable(codec-test "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/styx/test_codec.c")
target_link_libraries(codec-test PRIVATE inferno9)
add_test(NAME original-c-codec COMMAND codec-test)
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/native64.c")
    add_executable(server-native64-test "${INFERNO_SOURCE}/tools/libstyx/tests/native64.c")
    target_link_libraries(server-native64-test PRIVATE infernostyx)
    add_test(NAME original-c-server-native64 COMMAND server-native64-test)
endif()
add_executable(thread-state-test tests/thread-state.c)
target_link_libraries(thread-state-test PRIVATE inferno9)
add_test(NAME native-lib9-thread-state COMMAND thread-state-test)
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/stream.c")
    add_executable(server-stream-test "${INFERNO_SOURCE}/tools/libstyx/tests/stream.c")
    target_link_libraries(server-stream-test PRIVATE infernostyx)
    add_test(NAME original-c-server-stream COMMAND server-stream-test)
endif()
