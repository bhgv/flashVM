cmake_minimum_required(VERSION 3.10)
project(inferno_native_libraries VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS ON)
find_package(Threads REQUIRED)
find_package(PythonInterp 3 REQUIRED)
set(INFERNO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src" CACHE PATH "Exact Inferno source root")
set(INFERNO_HOST_SUPPORT "" CACHE PATH "Explicit shared bootstrap host support source root")
if(NOT EXISTS "${INFERNO_HOST_SUPPORT}/prepare_fmt.py")
    message(FATAL_ERROR "Set INFERNO_HOST_SUPPORT to the tools/limbo-bootstrap directory")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Initial native library acceptance requires Linux64")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(inferno_arch amd64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(inferno_arch arm64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(riscv64|rv64)$")
    set(inferno_arch riscv64)
else()
    message(FATAL_ERROR "Unsupported native64 architecture ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(inferno_headers "${INFERNO_SOURCE}/Linux/${inferno_arch}/include" "${INFERNO_SOURCE}/include")
add_subdirectory(lib9)
add_subdirectory(libstyx)
add_subdirectory(client)
enable_testing()
add_executable(codec-test "${CMAKE_CURRENT_SOURCE_DIR}/../runtime/styx/test_codec.c")
target_link_libraries(codec-test PRIVATE inferno9)
add_test(NAME original-c-codec COMMAND codec-test)
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/native64.c")
    add_executable(server-native64-test "${INFERNO_SOURCE}/tools/libstyx/tests/native64.c")
    target_link_libraries(server-native64-test PRIVATE infernostyx)
    add_test(NAME original-c-server-native64 COMMAND server-native64-test)
endif()
add_executable(thread-state-test tests/thread-state.c)
target_link_libraries(thread-state-test PRIVATE inferno9)
add_test(NAME native-lib9-thread-state COMMAND thread-state-test)
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/stream.c")
    add_executable(server-stream-test "${INFERNO_SOURCE}/tools/libstyx/tests/stream.c")
    target_link_libraries(server-stream-test PRIVATE infernostyx)
    add_test(NAME original-c-server-stream COMMAND server-stream-test)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/session.c")
    add_executable(server-session-test "${INFERNO_SOURCE}/tools/libstyx/tests/session.c")
    target_link_libraries(server-session-test PRIVATE infernostyx)
    add_test(NAME original-c-server-session COMMAND server-session-test)
    set_tests_properties(original-c-server-session PROPERTIES TIMEOUT 10)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/directories.c")
    add_executable(server-directories-test "${INFERNO_SOURCE}/tools/libstyx/tests/directories.c")
    target_link_libraries(server-directories-test PRIVATE infernostyx)
    add_test(NAME original-c-server-directories COMMAND server-directories-test)
    set_tests_properties(original-c-server-directories PROPERTIES TIMEOUT 10)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/lifecycle.c")
    add_executable(server-lifecycle-test "${INFERNO_SOURCE}/tools/libstyx/tests/lifecycle.c")
    target_link_libraries(server-lifecycle-test PRIVATE infernostyx)
    add_test(NAME original-c-server-lifecycle COMMAND server-lifecycle-test)
    set_tests_properties(original-c-server-lifecycle PROPERTIES TIMEOUT 10)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/loopback.c")
    add_executable(server-loopback-test "${INFERNO_SOURCE}/tools/libstyx/tests/loopback.c")
    target_link_libraries(server-loopback-test PRIVATE infernostyx)
    add_test(NAME original-c-server-loopback COMMAND "${PYTHON_EXECUTABLE}"
        "${INFERNO_SOURCE}/tools/libstyx/tests/loopback.py" "$<TARGET_FILE:server-loopback-test>")
    set_tests_properties(original-c-server-loopback PROPERTIES TIMEOUT 20)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/fid-context.c")
    add_executable(server-fid-context-test "${INFERNO_SOURCE}/tools/libstyx/tests/fid-context.c")
    target_link_libraries(server-fid-context-test PRIVATE infernostyx)
    add_test(NAME original-c-server-fid-context COMMAND server-fid-context-test)
    set_tests_properties(original-c-server-fid-context PROPERTIES TIMEOUT 10)
endif()
if(EXISTS "${INFERNO_SOURCE}/tools/libstyx/tests/deferred-attach.c")
    add_executable(server-deferred-attach-test "${INFERNO_SOURCE}/tools/libstyx/tests/deferred-attach.c")
    target_link_libraries(server-deferred-attach-test PRIVATE infernostyx)
    add_test(NAME original-c-server-deferred-attach COMMAND server-deferred-attach-test)
    set_tests_properties(original-c-server-deferred-attach PROPERTIES TIMEOUT 10)
endif()
set(INFERNO_VFS_SUPPORT "" CACHE PATH "Explicit flashVM repository root for native VFS file bridge")
if(INFERNO_VFS_SUPPORT)
    add_subdirectory(vfs)
endif()
