# Shared host adapters build independently of the unfinished native Dis host ABI.
cmake_minimum_required(VERSION 3.10)
project(disvm_shared_host VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(DVM_FLASHVM_SOURCE_ROOT "" CACHE PATH "Explicit FlashVM source root with include/ and ports/")
if(NOT EXISTS "${DVM_FLASHVM_SOURCE_ROOT}/ports/thread_port.h")
  message(FATAL_ERROR "Provide DVM_FLASHVM_SOURCE_ROOT; no sibling checkout or downloads are assumed")
endif()
get_filename_component(dvm_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(fvm "${DVM_FLASHVM_SOURCE_ROOT}")
# Use the very same implementations as FlashVM, without importing Inferno OS I/O.
if(WIN32)
  set(port windows)
elseif(ANDROID)
  set(port android)
elseif(APPLE)
  set(port mac)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(port linux)
else()
  message(FATAL_ERROR "No selected FlashVM host port for ${CMAKE_SYSTEM_NAME}")
endif()
message(STATUS "Shared FlashVM/Dis native host port: ${port}")
if(NOT TARGET disvm_module_image)
  add_library(disvm_module_image STATIC module_image.c)
endif()
target_include_directories(disvm_module_image PUBLIC "${dvm_root}/include")
add_library(disvm_flashvm_host STATIC flashvm_file.c flashvm_socket.c flashvm_channel.c flashvm_thread.c native_context.c)
target_include_directories(disvm_flashvm_host PUBLIC "${dvm_root}/include" "${fvm}/include" "${fvm}/ports")
target_link_libraries(disvm_flashvm_host PUBLIC disvm_module_image)
set(DVM_FLASHVM_TARGET "" CACHE STRING "Existing FlashVM CMake target, if embedded in one parent build")
if(DVM_FLASHVM_TARGET)
  if(NOT TARGET ${DVM_FLASHVM_TARGET})
    message(FATAL_ERROR "DVM_FLASHVM_TARGET must name an existing target")
  endif()
  target_link_libraries(disvm_flashvm_host PUBLIC ${DVM_FLASHVM_TARGET})
else()
  add_library(disvm_shared_ports STATIC "${fvm}/ports/file_port.c"
    "${fvm}/ports/${port}/thread.c" "${fvm}/ports/${port}/channel_sync.c"
    "${fvm}/ports/network/socket.c")
  target_include_directories(disvm_shared_ports PUBLIC "${fvm}/include" "${fvm}/ports")
  if(WIN32)
    target_link_libraries(disvm_shared_ports PUBLIC ws2_32)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(disvm_shared_ports PUBLIC Threads::Threads)
  endif()
  target_link_libraries(disvm_flashvm_host PUBLIC disvm_shared_ports)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(disvm_image_test "${dvm_root}/tests/module_image.c")
  target_link_libraries(disvm_image_test PRIVATE disvm_module_image)
  add_test(NAME disvm_module_images COMMAND disvm_image_test)
  add_executable(disvm_host_test "${dvm_root}/tests/shared_host.c")
  target_link_libraries(disvm_host_test PRIVATE disvm_flashvm_host)
  add_test(NAME disvm_shared_native_host COMMAND disvm_host_test)
  set_tests_properties(disvm_module_images disvm_shared_native_host PROPERTIES TIMEOUT 20)
endif()

# Channel acceptance must link the real FlashVM VM, not a test replacement.
set(DVM_FLASHVM_LIBRARIES "" CACHE STRING "Explicit FlashVM library/dependency list for channel integration test")
if(BUILD_TESTING AND DVM_FLASHVM_LIBRARIES)
  enable_language(CXX)
  add_executable(disvm_channel_test "${dvm_root}/tests/module_channel.c")
  set_target_properties(disvm_channel_test PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(disvm_channel_test PRIVATE disvm_flashvm_host ${DVM_FLASHVM_LIBRARIES})
  add_executable(disvm_module_image_example "${dvm_root}/examples/module-image/main.c")
  set_target_properties(disvm_module_image_example PROPERTIES LINKER_LANGUAGE CXX)
  target_link_libraries(disvm_module_image_example PRIVATE disvm_flashvm_host ${DVM_FLASHVM_LIBRARIES})
  add_test(NAME disvm_real_flashvm_channel COMMAND disvm_channel_test)
  set_tests_properties(disvm_real_flashvm_channel PROPERTIES TIMEOUT 20)
endif()
if(BUILD_TESTING)
  find_package(PythonInterp 3 QUIET)
  if(PYTHONINTERP_FOUND)
    add_executable(disvm_socket_test "${dvm_root}/tests/module_socket.c")
    target_link_libraries(disvm_socket_test PRIVATE disvm_flashvm_host)
    add_test(NAME disvm_real_flashvm_socket COMMAND "${PYTHON_EXECUTABLE}"
      "${dvm_root}/tests/module_socket.py" $<TARGET_FILE:disvm_socket_test>)
    set_tests_properties(disvm_real_flashvm_socket PROPERTIES TIMEOUT 20)
  endif()
endif()

if(BUILD_TESTING AND PYTHONINTERP_FOUND AND CMAKE_SYSTEM_NAME STREQUAL "Linux"
    AND NOT ANDROID AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME disvm_native_thread_registers COMMAND "${PYTHON_EXECUTABLE}"
    "${dvm_root}/tests/native_registers.py" --flashvm-root "${fvm}")
  set_tests_properties(disvm_native_thread_registers PROPERTIES TIMEOUT 30
    ENVIRONMENT "CC=${CMAKE_C_COMPILER}")
endif()
