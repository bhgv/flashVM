# Independent C99 library; matching compiler/linker dependencies are explicit.
cmake_minimum_required(VERSION 3.10)
project(disvm_compiler_service VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(DVM_COMPILER_FIR_INCLUDE "" CACHE PATH "Matching FIR public headers")
set(DVM_COMPILER_FIR_LIB "" CACHE PATH "Matching FIR static library directory")
set(DVM_COMPILER_SWL_INCLUDE "" CACHE PATH "Matching SWL public headers")
set(DVM_COMPILER_SWL_LIB "" CACHE PATH "Matching SWL static library directory")
foreach(header "${DVM_COMPILER_FIR_INCLUDE}/fir.h" "${DVM_COMPILER_SWL_INCLUDE}/swl.h")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Compiler service requires matching explicit headers: ${header}")
  endif()
endforeach()
set(compiler_libraries)
foreach(unit fir1_source_api fir1_ir fir1_io)
  if(NOT EXISTS "${DVM_COMPILER_FIR_LIB}/lib${unit}.a")
    message(FATAL_ERROR "Missing compiler library: ${unit}")
  endif()
  list(APPEND compiler_libraries "${DVM_COMPILER_FIR_LIB}/lib${unit}.a")
endforeach()
if(NOT EXISTS "${DVM_COMPILER_SWL_LIB}/libswl_core.a")
  message(FATAL_ERROR "Missing linker library: swl_core")
endif()
list(APPEND compiler_libraries "${DVM_COMPILER_SWL_LIB}/libswl_core.a")
add_library(disvm_compiler_engine STATIC engine.c)
target_include_directories(disvm_compiler_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${DVM_COMPILER_FIR_INCLUDE}" "${DVM_COMPILER_SWL_INCLUDE}")
target_link_libraries(disvm_compiler_engine PUBLIC ${compiler_libraries} m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(disvm_compiler_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(DVM_COMPILER_TESTS "Build standalone compiler service API checks" OFF)
if(DVM_COMPILER_TESTS)
  enable_testing()
  add_executable(disvm_compiler_api test_engine.c)
  target_link_libraries(disvm_compiler_api PRIVATE disvm_compiler_engine)
  add_test(NAME disvm_compiler_api COMMAND disvm_compiler_api)
  add_test(NAME disvm_compiler_unavailable_temp COMMAND "${CMAKE_COMMAND}" -E env
    TMPDIR=${CMAKE_CURRENT_BINARY_DIR}/absent-directory $<TARGET_FILE:disvm_compiler_api> --temp-failure)
endif()
