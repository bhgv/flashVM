# Optional VM host application. SDL is a host presentation SDK, not a bundled VM dependency.
find_package(SDL2 2.0.5 CONFIG REQUIRED)
if(ANDROID)
  message(FATAL_ERROR "Desktop frontend is not an Android Activity; build the VM without this option")
elseif(WIN32)
  set(frontend_os windows)
elseif(APPLE)
  set(frontend_os macos)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(frontend_os linux)
else()
  message(FATAL_ERROR "Desktop frontend currently supports Linux, Windows and macOS")
endif()
add_executable(flashvm_player ${frontend_os}/main.c common/player.c common/options.c common/sdl_window.c)
target_link_libraries(flashvm_player PRIVATE flashvm SDL2::SDL2)
if(WIN32)
  target_link_libraries(flashvm_player PRIVATE shell32)
endif()
set_target_properties(flashvm_player PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
if(FVM_BUILD_TESTS)
  add_executable(flashvm_frontend_options tests/options.c common/options.c)
  target_include_directories(flashvm_frontend_options PRIVATE ../include)
  set_target_properties(flashvm_frontend_options PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(flashvm_frontend_options PRIVATE /UNDEBUG)
  else()
    target_compile_options(flashvm_frontend_options PRIVATE -UNDEBUG)
  endif()
  add_test(NAME flashvm_frontend_options COMMAND flashvm_frontend_options)
  add_test(NAME flashvm_player_help COMMAND flashvm_player --help)
endif()

# Native display tests are opt-in: CI without a display must not silently count dummy-video runs.
option(FVM_DESKTOP_WINDOW_TESTS "Run real desktop window/presentation smoke tests" OFF)
if(FVM_BUILD_TESTS AND FVM_DESKTOP_WINDOW_TESTS AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(flashvm_frontend_window_test tests/window.c)
  target_link_libraries(flashvm_frontend_window_test PRIVATE flashvm SDL2::SDL2)
  if(MSVC)
    target_compile_options(flashvm_frontend_window_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(flashvm_frontend_window_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME flashvm_desktop_window_pixels COMMAND flashvm_frontend_window_test)
  add_test(NAME flashvm_desktop_playback COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE:flashvm_player>)
  set_tests_properties(flashvm_desktop_window_pixels flashvm_desktop_playback PROPERTIES TIMEOUT 60)
endif()
