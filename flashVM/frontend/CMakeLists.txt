# Optional VM host application. SDL is a host presentation SDK, not a bundled VM dependency.
find_package(SDL2 2.0.9 CONFIG REQUIRED)
if(ANDROID)
  set(frontend_os android)
elseif(WIN32)
  set(frontend_os windows)
elseif(APPLE)
  set(frontend_os macos)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(frontend_os linux)
else()
  message(FATAL_ERROR "Desktop frontend currently supports Linux, Windows and macOS")
endif()
set(frontend_sources ${frontend_os}/main.c common/player.c common/options.c common/sdl_window.c common/audio.c common/audio_trace.c)
if(ANDROID)
  add_library(flashvm_player SHARED ${frontend_sources})
  set_target_properties(flashvm_player PROPERTIES OUTPUT_NAME flashvm_android)
else()
  add_executable(flashvm_player ${frontend_sources})
endif()
target_link_libraries(flashvm_player PRIVATE flashvm SDL2::SDL2)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID)
  # Portable binary packages put the SDL runtime next to the player in lib/.
  set_target_properties(flashvm_player PROPERTIES BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN/lib")
endif()
if(WIN32)
  target_link_libraries(flashvm_player PRIVATE shell32)
endif()
set_target_properties(flashvm_player PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
if(FVM_BUILD_TESTS AND NOT ANDROID)
  add_executable(flashvm_frontend_options tests/options.c common/options.c)
  target_include_directories(flashvm_frontend_options PRIVATE ../include)
  set_target_properties(flashvm_frontend_options PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(flashvm_frontend_options PRIVATE /UNDEBUG)
  else()
    target_compile_options(flashvm_frontend_options PRIVATE -UNDEBUG)
  endif()
  add_executable(flashvm_frontend_keyboard tests/keyboard.c)
  target_link_libraries(flashvm_frontend_keyboard PRIVATE flashvm SDL2::SDL2)
  set_target_properties(flashvm_frontend_keyboard PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(flashvm_frontend_keyboard PRIVATE /UNDEBUG)
  else()
    target_compile_options(flashvm_frontend_keyboard PRIVATE -UNDEBUG)
  endif()
  add_test(NAME flashvm_frontend_keyboard COMMAND flashvm_frontend_keyboard)
  add_test(NAME flashvm_frontend_options COMMAND flashvm_frontend_options)
  add_test(NAME flashvm_player_help COMMAND flashvm_player --help)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME flashvm_player_audio_assets COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/audio_assets.py)
endif()

# Explicit Linux X11 input/pixel acceptance for the delivered demo files; requires Pillow/xdotool.
option(FVM_DESKTOP_DEMO_TESTS "Run X11 input/pixel demo acceptance with real SDL and dummy audio" OFF)
if(FVM_BUILD_TESTS AND FVM_DESKTOP_DEMO_TESTS AND NOT CMAKE_CROSSCOMPILING AND NOT ANDROID)
  if(NOT FVM_ENABLE_SDL_AUDIO OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Demo input acceptance requires Linux X11 and FVM_ENABLE_SDL_AUDIO")
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(FVM_XDOTOOL xdotool)
  if(NOT FVM_XDOTOOL)
    message(FATAL_ERROR "FVM_DESKTOP_DEMO_TESTS requires xdotool (or set FVM_XDOTOOL)")
  endif()
  add_test(NAME flashvm_player_demo_movies COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/demo_movies.py $<TARGET_FILE:flashvm_player> ${FVM_XDOTOOL})
  set_tests_properties(flashvm_player_demo_movies PROPERTIES TIMEOUT 60)
endif()
