  if(FVM_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    find_program(FVM_OPENSSL_PROGRAM openssl)
  endif()
  find_package(CURL 7.62 REQUIRED)
  add_library(flashvm_http_transport STATIC common/http.c common/http_redirect.c common/socket_url.c
    ../ports/network/clock.c)
  target_include_directories(flashvm_http_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common ${PROJECT_SOURCE_DIR}/include)
  target_include_directories(flashvm_http_transport PRIVATE ${CURL_INCLUDE_DIRS})
  if(TARGET CURL::libcurl)
    target_link_libraries(flashvm_http_transport PUBLIC CURL::libcurl)
  else()
    target_link_libraries(flashvm_http_transport PUBLIC ${CURL_LIBRARIES})
  endif()
  if(FVM_PLAYER_HOST_SHARED OR ANDROID)
    set_target_properties(flashvm_http_transport PROPERTIES POSITION_INDEPENDENT_CODE YES)
  endif()
  # Protocol fixtures also compile player.c directly; use the same optional provider.
  add_definitions(-DFVM_DESKTOP_HTTP)
  link_libraries(flashvm_http_transport)
