# The requested fork is the default Clutter provider; system SDKs are explicit test-only opt-ins.
set(FVM_CLUTTER3D_SDK "" CACHE PATH "SDK prefix built from bhgv's Clutter/Cogl fork")
option(FVM_CLUTTER3D_SYSTEM_TEST_SDK "Allow a system Clutter SDK for auxiliary tests only" OFF)
if(FVM_EXAMPLE_CLUTTER3D)
  if(NOT FVM_CLUTTER3D_SDK AND NOT FVM_CLUTTER3D_SYSTEM_TEST_SDK)
    message(FATAL_ERROR "Set FVM_CLUTTER3D_SDK to the prepared fork SDK; see external/clutter3d/README.md")
  endif()
  if(FVM_CLUTTER3D_SDK)
    if(NOT EXISTS "${FVM_CLUTTER3D_SDK}/clutter3d-source.json")
      message(FATAL_ERROR "Clutter SDK has no source provenance marker; run build_linux.sh first")
    endif()
    file(READ "${FVM_CLUTTER3D_SDK}/clutter3d-source.json" sdk_provenance)
    string(FIND "${sdk_provenance}" "2a616b264f0045a3ba00ce77aa1930ca905956ea" sdk_revision)
    if(sdk_revision LESS 0)
      message(FATAL_ERROR "Clutter SDK does not identify the selected repository revision")
    endif()
    file(SHA256 "${PROJECT_SOURCE_DIR}/external/clutter3d/patches/actor-subpixel.patch" actor_patch_hash)
    string(FIND "${sdk_provenance}" "${actor_patch_hash}" sdk_actor_patch)
    if(sdk_actor_patch LESS 0)
      message(FATAL_ERROR "Rebuild the fork SDK with the v0.601 subpixel actor fix")
    endif()
    set(saved_pkg_path "$ENV{PKG_CONFIG_PATH}")
    set(ENV{PKG_CONFIG_PATH} "${FVM_CLUTTER3D_SDK}/lib/pkgconfig:${saved_pkg_path}")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FVM_CLUTTER REQUIRED IMPORTED_TARGET clutter-1.0>=1.12)
  if(FVM_CLUTTER3D_SDK)
    pkg_get_variable(clutter_actual_prefix clutter-1.0 prefix)
    get_filename_component(clutter_actual_prefix "${clutter_actual_prefix}" REALPATH)
    get_filename_component(clutter_expected_prefix "${FVM_CLUTTER3D_SDK}" REALPATH)
    if(NOT clutter_actual_prefix STREQUAL clutter_expected_prefix)
      message(FATAL_ERROR "pkg-config selected another Clutter SDK; configure a fresh build directory")
    endif()
    foreach(cogl_package cogl-1.0 cogl-pango-1.0)
      pkg_get_variable(cogl_actual_prefix ${cogl_package} prefix)
      get_filename_component(cogl_actual_prefix "${cogl_actual_prefix}" REALPATH)
      if(NOT cogl_actual_prefix STREQUAL clutter_expected_prefix)
        message(FATAL_ERROR "${cogl_package} is not from the same fork SDK; refusing mixed providers")
      endif()
    endforeach()
    set(ENV{PKG_CONFIG_PATH} "${saved_pkg_path}")
  endif()
  add_executable(flashvm_example_clutter3d clutter3d/main.c)
  target_link_libraries(flashvm_example_clutter3d PRIVATE flashvm PkgConfig::FVM_CLUTTER)
  if(FVM_CLUTTER3D_SDK)
    # Static Clutter/Cogl need the SDK's explicit private/transitive dependency closure.
    target_link_libraries(flashvm_example_clutter3d PRIVATE ${FVM_CLUTTER_STATIC_LDFLAGS})
  endif()
endif()
