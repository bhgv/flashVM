# Explicit roots make this example usable with detached renamed components.
cmake_minimum_required(VERSION 3.10)
project(swo_ide_example LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(FIR_ROOT "" CACHE PATH "Explicit FIR v1 component root")
set(SWL_ROOT "" CACHE PATH "Explicit SWL component root")
set(FRONTENDS_ROOT "" CACHE PATH "Optional syntax frontends component root")
if(NOT EXISTS "${FIR_ROOT}/include/fir.h" OR NOT EXISTS "${SWL_ROOT}/include/swl.h")
  message(FATAL_ERROR "Set FIR_ROOT and SWL_ROOT explicitly")
endif()
include(CTest)
set(ide_testing ${BUILD_TESTING})
set(BUILD_TESTING OFF)
add_subdirectory("${FIR_ROOT}" fir)
add_subdirectory("${SWL_ROOT}" swl)
if(FRONTENDS_ROOT)
  add_subdirectory("${FRONTENDS_ROOT}" frontends)
endif()
set(BUILD_TESTING ${ide_testing})
add_executable(ide_pipeline main.c)
target_link_libraries(ide_pipeline PRIVATE fir1_ir fir1_source_api swl_core)
if(FRONTENDS_ROOT)
  target_include_directories(ide_pipeline PRIVATE "${FRONTENDS_ROOT}")
  target_compile_definitions(ide_pipeline PRIVATE IDE_WITH_FRONTENDS)
  target_link_libraries(ide_pipeline PRIVATE as1_frontend as2_frontend as3_frontend limbo_frontend)
endif()
if(BUILD_TESTING)
  add_test(NAME ide_pipeline COMMAND ide_pipeline "${CMAKE_CURRENT_BINARY_DIR}/ide.swf")
endif()
