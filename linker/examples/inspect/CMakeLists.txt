# Standalone host: discover the linker through an explicit SDK/source location.
cmake_minimum_required(VERSION 3.16)
project(swl_inspect_example LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SWL_ROOT "" CACHE PATH "Explicit standalone SWL source directory")
if(NOT EXISTS "${SWL_ROOT}/include/swl.h")
  message(FATAL_ERROR "Set SWL_ROOT to the independently buildable linker project")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Do not execute SDK tests from this host" FORCE)
add_subdirectory("${SWL_ROOT}" swl-sdk)
add_executable(inspect_example main.c)
target_link_libraries(inspect_example PRIVATE swl_core)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(inspect_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
