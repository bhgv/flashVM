# Autonomous C99 linker: no compiler, Coco/R or VM dependency.
cmake_minimum_required(VERSION 3.10)
project(swf_linker VERSION 0.3.14 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(swl_core src/linker.c src/archive.c src/assets.c src/inspect.c)
target_include_directories(swl_core PUBLIC include)
add_executable(swl src/main.c)
target_link_libraries(swl PRIVATE swl_core)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(swl_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(swl PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  # FindPython3 starts at CMake 3.12; retain standalone 3.10 builds.
  if(CMAKE_VERSION VERSION_LESS "3.12")
    find_package(PythonInterp 3 REQUIRED)
    set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
    set(Python3_Interpreter_FOUND "${PYTHONINTERP_FOUND}")
  else()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
  endif()
  add_test(NAME swl_formats COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.py $<TARGET_FILE:swl>)
endif()
if(BUILD_TESTING)
  add_executable(swl_embed tests/embed.c)
  target_link_libraries(swl_embed PRIVATE swl_core)
  add_test(NAME swl_embedding COMMAND swl_embed)
endif()
if(BUILD_TESTING)
  add_executable(swl_inspection tests/inspect.c)
  target_link_libraries(swl_inspection PRIVATE swl_core)
  add_test(NAME swl_inspection COMMAND swl_inspection)
endif()
