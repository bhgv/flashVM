# Optional compiler-only orchestration; each language/frontend builds alone too.
cmake_minimum_required(VERSION 3.10)
project(flash_compilers LANGUAGES C)
# Apply the same C99 contract to orchestration-owned embedding hosts.
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_subdirectory(as1)
add_subdirectory(as2)
add_subdirectory(as3)
add_subdirectory(limbo)

if(BUILD_TESTING)
  add_executable(compilers_embed tests/embed_all.c)
  target_include_directories(compilers_embed PRIVATE .)
  target_link_libraries(compilers_embed PRIVATE as1_frontend as2_frontend as3_frontend limbo_frontend)
  add_test(NAME compilers_embedding COMMAND compilers_embed)
endif()

# Optional explicit dependency: detached AS/Limbo projects never assume a sibling FIR checkout.
set(FIR_SYNTAX_ROOT "" CACHE PATH "Independent FIR structural frontend directory (with firsyn.atg)")
if(FIR_SYNTAX_ROOT)
  if(NOT EXISTS "${FIR_SYNTAX_ROOT}/firsyn.atg" OR NOT EXISTS "${FIR_SYNTAX_ROOT}/firsyn.h")
    message(FATAL_ERROR "FIR_SYNTAX_ROOT must name the independent FIR frontend directory")
  endif()
  if(NOT TARGET firsyn_frontend)
    add_subdirectory("${FIR_SYNTAX_ROOT}" fir-syntax)
  endif()
  if(BUILD_TESTING)
    add_executable(compilers_embed_five tests/embed_five.c)
    target_include_directories(compilers_embed_five PRIVATE .)
    target_link_libraries(compilers_embed_five PRIVATE
      as1_frontend as2_frontend as3_frontend limbo_frontend firsyn_frontend)
    add_test(NAME compilers_five_embedding COMMAND compilers_embed_five)
  endif()
endif()
