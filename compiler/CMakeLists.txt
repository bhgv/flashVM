# Optional compiler-only orchestration; each language/frontend builds alone too.
cmake_minimum_required(VERSION 3.16)
project(flash_compilers LANGUAGES C)
include(CTest)
add_subdirectory(as1)
add_subdirectory(as2)
add_subdirectory(as3)
add_subdirectory(limbo)

if(BUILD_TESTING)
  add_executable(compilers_embed tests/embed_all.c)
  target_include_directories(compilers_embed PRIVATE .)
  target_link_libraries(compilers_embed PRIVATE as1_frontend as2_frontend as3_frontend limbo_frontend)
  add_test(NAME compilers_embedding COMMAND compilers_embed)
endif()
