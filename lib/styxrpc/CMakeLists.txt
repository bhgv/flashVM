# EN: Independent C RPC codec and callback endpoint over installed communication libraries.
# RU: Независимый C-кодек RPC и колбэки поверх установленных библиотек связи.
cmake_minimum_required(VERSION 3.16)
project(StyxRpc VERSION 0.1.0 LANGUAGES C)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Unibus CONFIG REQUIRED)
option(STYXRPC_HOSTED "Build the Styx launch and virtual tree adapters" OFF)
option(STYXRPC_EXAMPLES "Build the hosted Styx callback device" OFF)
add_library(styxrpc STATIC src/rpc.c src/managed.c src/graph.c src/channel.c src/channel_call.c src/channel_fid.c src/interface.c src/vfs.c protocol/call_record.c protocol/call_scope.c
    protocol/load_profile.c protocol/ref_pair_record.c protocol/scalar_record.c
    protocol/scalar_verify.c protocol/tuple_record.c protocol/utf8.c protocol/variable_file.c protocol/indexed_variable.c)
target_link_libraries(styxrpc PUBLIC Unibus::wire)
target_include_directories(styxrpc PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>" PRIVATE protocol include/styxrpc)
set_target_properties(styxrpc PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO EXPORT_NAME rpc)
add_library(StyxRpc::rpc ALIAS styxrpc)
set(styxrpc_targets styxrpc)
if(STYXRPC_HOSTED OR STYXRPC_EXAMPLES)
    find_package(Styx CONFIG REQUIRED)
    set(STYXRPC_HAS_LAUNCH ON)
    add_library(styxrpc_launch STATIC src/launch.c src/channel_exchange.c)
    target_include_directories(styxrpc_launch PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
    target_link_libraries(styxrpc_launch PUBLIC Styx::client StyxRpc::rpc)
    set_target_properties(styxrpc_launch PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES EXPORT_NAME launch)
    add_library(StyxRpc::launch ALIAS styxrpc_launch)
    add_library(styxrpc_serve STATIC src/serve.c src/program.c src/collection.c)
    target_link_libraries(styxrpc_serve PUBLIC StyxRpc::rpc Styx::phy_server)
    set_target_properties(styxrpc_serve PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES EXPORT_NAME serve)
    add_library(StyxRpc::serve ALIAS styxrpc_serve)
    list(APPEND styxrpc_targets styxrpc_launch styxrpc_serve)
endif()
if(STYXRPC_EXAMPLES)
    add_executable(styxrpc-launch examples/disvm/launch.c)
    target_link_libraries(styxrpc-launch PRIVATE StyxRpc::launch Styx::bidir)
    add_executable(styxrpc-named examples/disvm/named.c examples/disvm/reverse.c)
    target_link_libraries(styxrpc-named PRIVATE StyxRpc::launch StyxRpc::serve Styx::bidir)
    add_executable(styxrpc-device examples/device.c examples/adjust.c examples/graph_device.c)
    target_include_directories(styxrpc-device PRIVATE protocol)
    target_link_libraries(styxrpc-device PRIVATE styxrpc Styx::phy_server)
    add_executable(styxrpc-interface examples/disvm/interface.c examples/adjust.c examples/graph_device.c)
    target_include_directories(styxrpc-interface PRIVATE protocol)
    target_link_libraries(styxrpc-interface PRIVATE styxrpc)
    add_executable(styxrpc-channel-device examples/disvm/channel_device.c examples/disvm/owned_port.c)
    target_include_directories(styxrpc-channel-device PRIVATE protocol)
    target_link_libraries(styxrpc-channel-device PRIVATE StyxRpc::launch StyxRpc::serve Styx::bidir)
    add_executable(styxrpc-client examples/client.c)
    target_link_libraries(styxrpc-client PRIVATE styxrpc Styx::client)
    if(BUILD_TESTING)
        find_package(Python3 QUIET COMPONENTS Interpreter)
        if(Python3_Interpreter_FOUND)
            add_test(NAME styxrpc-native-socket COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/examples/socketpair.py"
                "$<TARGET_FILE:styxrpc-device>" "$<TARGET_FILE:styxrpc-client>")
        endif()
    endif()
endif()
if(BUILD_TESTING)
    if(TARGET styxrpc_serve)
        add_executable(styxrpc-directory-test tests/directory.c)
        target_link_libraries(styxrpc-directory-test PRIVATE StyxRpc::serve Styx::client)
        add_test(NAME styxrpc-live-directory COMMAND styxrpc-directory-test)
        add_executable(styxrpc-channel-test tests/channel_socket.c)
        target_link_libraries(styxrpc-channel-test PRIVATE StyxRpc::serve Styx::client)
        add_test(NAME styxrpc-unibus-socket COMMAND styxrpc-channel-test)
        set_tests_properties(styxrpc-unibus-socket PROPERTIES TIMEOUT 10)
        set_tests_properties(styxrpc-live-directory PROPERTIES TIMEOUT 10)
    endif()
    add_executable(styxrpc-test tests/scalar.c)
    target_include_directories(styxrpc-test PRIVATE protocol)
    target_link_libraries(styxrpc-test PRIVATE styxrpc)
    add_test(NAME styxrpc-scalar COMMAND styxrpc-test)
    add_executable(styxrpc-managed-test tests/managed.c)
    target_include_directories(styxrpc-managed-test PRIVATE protocol)
    target_link_libraries(styxrpc-managed-test PRIVATE styxrpc)
    add_test(NAME styxrpc-managed COMMAND styxrpc-managed-test)
    add_executable(styxrpc-records-test tests/portable_records.c)
    target_include_directories(styxrpc-records-test PRIVATE protocol)
    target_link_libraries(styxrpc-records-test PRIVATE styxrpc)
    add_test(NAME styxrpc-portable-records COMMAND styxrpc-records-test)
endif()
include(cmake/OnlyC.cmake)
require_c_libraries(${styxrpc_targets})
install(TARGETS ${styxrpc_targets} EXPORT StyxRpcTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT StyxRpcTargets NAMESPACE StyxRpc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StyxRpc)
configure_package_config_file(cmake/StyxRpcConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/StyxRpcConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StyxRpc)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/StyxRpcConfigVersion.cmake"
    VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/StyxRpcConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/StyxRpcConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StyxRpc)
