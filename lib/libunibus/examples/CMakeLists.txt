add_executable(unibus-nested nested.c)
target_link_libraries(unibus-nested PRIVATE Unibus::wire)
set_target_properties(unibus-nested PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(unibus-nested PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME unibus-nested COMMAND unibus-nested)
if(UNIBUS_QUEUES)
    add_executable(unibus-queues queues.c)
    target_link_libraries(unibus-queues PRIVATE Unibus::queues)
    set_target_properties(unibus-queues PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    target_compile_definitions(unibus-queues PRIVATE _POSIX_C_SOURCE=200809L)
    add_test(NAME unibus-queues COMMAND unibus-queues)
    set_tests_properties(unibus-queues PROPERTIES TIMEOUT 10)
    foreach(check codec queues rollover fault)
        add_executable(unibus-order-${check} ../order/tests/${check}.c)
        target_link_libraries(unibus-order-${check} PRIVATE Unibus::queues)
        set_target_properties(unibus-order-${check} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
        if(check STREQUAL "fault")
            target_link_options(unibus-order-${check} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc)
        endif()
        add_test(NAME unibus-order-${check} COMMAND unibus-order-${check})
        set_tests_properties(unibus-order-${check} PROPERTIES TIMEOUT 15)
    endforeach()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        # GNU/LLVM linker wrapping injects failures into the actual example, not a duplicate implementation.
        add_library(unibus-failure-example OBJECT queues.c)
        target_link_libraries(unibus-failure-example PRIVATE Unibus::queues)
        target_compile_definitions(unibus-failure-example PRIVATE main=queue_example_main _POSIX_C_SOURCE=200809L)
        set_target_properties(unibus-failure-example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
        add_executable(unibus-example-failures ../tests/queue_failures.c $<TARGET_OBJECTS:unibus-failure-example>)
        target_link_libraries(unibus-example-failures PRIVATE Unibus::queues)
        target_link_options(unibus-example-failures PRIVATE -Wl,--wrap=styxordercreate -Wl,--wrap=styxorderfree
            -Wl,--wrap=pthread_barrier_init -Wl,--wrap=pthread_barrier_destroy -Wl,--wrap=pthread_create)
        set_target_properties(unibus-example-failures PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
        add_test(NAME unibus-example-failures COMMAND unibus-example-failures)
        set_tests_properties(unibus-example-failures PROPERTIES TIMEOUT 10)
    endif()
    if(UNIX)
        set(bridge_core bridge/worker.c bridge/io.c bridge/connection.c bridge/lifecycle.c bridge/receive.c)
        add_executable(unibus-9p-bridge bridge/main.c ${bridge_core})
        target_link_libraries(unibus-9p-bridge PRIVATE Unibus::wire Threads::Threads)
        target_compile_options(unibus-9p-bridge PRIVATE -Wall -Wextra -Werror -Wpedantic)
        set_target_properties(unibus-9p-bridge PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME unibus-gateway COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bridge/check.py" "$<TARGET_FILE:unibus-9p-bridge>")
        set_tests_properties(unibus-gateway PROPERTIES TIMEOUT 30)
        add_test(NAME unibus-gateway-edges COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/gateway_edges.py" "$<TARGET_FILE:unibus-9p-bridge>")
        set_tests_properties(unibus-gateway-edges PROPERTIES TIMEOUT 60)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_executable(unibus-bridge-lifecycle ../tests/bridge_lifecycle.c ${bridge_core})
            target_include_directories(unibus-bridge-lifecycle PRIVATE bridge)
            target_link_libraries(unibus-bridge-lifecycle PRIVATE Unibus::wire Threads::Threads)
            target_link_options(unibus-bridge-lifecycle PRIVATE -Wl,--wrap=pthread_mutex_init -Wl,--wrap=pthread_mutex_destroy
                -Wl,--wrap=pthread_cond_init -Wl,--wrap=pthread_cond_destroy -Wl,--wrap=pthread_create)
            set_target_properties(unibus-bridge-lifecycle PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
            add_test(NAME unibus-bridge-lifecycle COMMAND unibus-bridge-lifecycle)
            set_tests_properties(unibus-bridge-lifecycle PROPERTIES TIMEOUT 10)
            add_executable(unibus-bridge-io ../tests/bridge_io.c bridge/io.c)
            target_include_directories(unibus-bridge-io PRIVATE bridge)
            target_link_libraries(unibus-bridge-io PRIVATE Unibus::wire Threads::Threads)
            target_link_options(unibus-bridge-io PRIVATE -Wl,--wrap=read -Wl,--wrap=write)
            set_target_properties(unibus-bridge-io PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
            add_test(NAME unibus-bridge-io COMMAND unibus-bridge-io)
            set_tests_properties(unibus-bridge-io PROPERTIES TIMEOUT 10)
        endif()
    endif()
endif()
