# Optional portable host-side parallel drawing jobs (C89).
option(FVM_ENABLE_RENDER_JOBS "Enable portable render job worker threads" ON)
if(FVM_ENABLE_RENDER_JOBS)
  if(WIN32)
    set(FVM_THREAD_SOURCE ports/windows/thread.c)
  elseif(ANDROID)
    set(FVM_THREAD_SOURCE ports/android/thread.c)
  elseif(APPLE)
    set(FVM_THREAD_SOURCE ports/mac/thread.c)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(FVM_THREAD_SOURCE ports/linux/thread.c)
  else()
    message(FATAL_ERROR "Render jobs require a supported port or FVM_ENABLE_RENDER_JOBS=OFF")
  endif()
  target_sources(flashvm PRIVATE src/render/render_jobs.c ${FVM_THREAD_SOURCE})
  if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(flashvm ${CMAKE_THREAD_LIBS_INIT})
  endif()
  if(FVM_BUILD_TESTS)
    add_executable(flashvm_render_jobs tests/render_jobs.c)
    target_link_libraries(flashvm_render_jobs flashvm)
    add_test(NAME flashvm_render_jobs COMMAND flashvm_render_jobs)
  endif()
  if(FVM_BUILD_EXAMPLES)
    add_executable(flashvm_example_parallel_render examples/embedding/parallel_render/main.c)
    target_link_libraries(flashvm_example_parallel_render flashvm)
    if(FVM_BUILD_TESTS)
      add_test(NAME flashvm_example_parallel_render COMMAND flashvm_example_parallel_render)
    endif()
  endif()
endif()

# Native OS input and TCP socket adapters. The VM core remains OS-agnostic.
option(FVM_ENABLE_OS_PORTS "Build native OS input and TCP adapters" ON)
if(FVM_ENABLE_OS_PORTS)
  if(WIN32)
    set(FVM_OS_PORT windows)
  elseif(ANDROID)
    set(FVM_OS_PORT android)
  elseif(APPLE)
    set(FVM_OS_PORT mac)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(FVM_OS_PORT linux)
  else()
    message(FATAL_ERROR "Unsupported OS ports; set FVM_ENABLE_OS_PORTS=OFF")
  endif()
  target_sources(flashvm PRIVATE ports/${FVM_OS_PORT}/input.c ports/${FVM_OS_PORT}/network.c)
  target_include_directories(flashvm PUBLIC ${PROJECT_SOURCE_DIR}/ports)
  if(WIN32)
    target_link_libraries(flashvm ws2_32)
  elseif(APPLE)
    find_library(FVM_APPLICATION_SERVICES ApplicationServices REQUIRED)
    target_link_libraries(flashvm ${FVM_APPLICATION_SERVICES})
  elseif(ANDROID)
    target_link_libraries(flashvm android)
  endif()
  if(FVM_BUILD_TESTS)
    add_executable(flashvm_os_ports tests/os_ports.c)
    target_link_libraries(flashvm_os_ports flashvm)
    add_test(NAME flashvm_os_ports COMMAND flashvm_os_ports)
  endif()
  if(FVM_BUILD_EXAMPLES)
    add_executable(flashvm_embedding_os_ports examples/embedding/os_ports/main.c)
    target_link_libraries(flashvm_embedding_os_ports flashvm)
    add_test(NAME flashvm_embedding_os_ports COMMAND flashvm_embedding_os_ports)
  endif()
endif()
