cmake_minimum_required(VERSION 2.8.12)
project(FlashVM C CXX)
option(FVM_BUILD_TESTS "Build tests" ON)
option(FVM_ENABLE_JIT "Build JIT components when available" OFF)
option(FVM_ENABLE_RENDER "Build rendering components when available" OFF)
option(FVM_ENABLE_MEDIA "Build media components when available" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++98 -Wall -Wextra -pedantic")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=c89 -Wall -Wextra -pedantic")
endif()
include_directories(${PROJECT_SOURCE_DIR}/include)
add_library(flashvm STATIC src/core/runtime.cpp src/capi/flashvm_c.cpp src/capi/value_c.cpp src/swf/parser.cpp src/swf/swf_c.cpp src/avm1/decoder.cpp src/avm1/interpreter.cpp)
if(FVM_BUILD_TESTS)
  enable_testing()
  add_executable(flashvm_smoke tests/smoke.c)
  target_link_libraries(flashvm_smoke flashvm)
  add_test(NAME flashvm_smoke COMMAND flashvm_smoke)
endif()
if(FVM_BUILD_TESTS)
  add_executable(flashvm_objects tests/objects.c)
  target_link_libraries(flashvm_objects flashvm)
  add_test(NAME flashvm_objects COMMAND flashvm_objects)
endif()

if(FVM_BUILD_TESTS)
 add_executable(flashvm_swf tests/swf.c)
 target_link_libraries(flashvm_swf flashvm)
 add_test(NAME flashvm_swf COMMAND flashvm_swf)
endif()

if(FVM_BUILD_TESTS)
 add_executable(flashvm_avm1 tests/avm1.c)
 target_link_libraries(flashvm_avm1 flashvm)
 add_test(NAME flashvm_avm1 COMMAND flashvm_avm1)
endif()

option(FVM_BUILD_EXAMPLES "Build C89 embedding and extension examples" ON)
if(FVM_BUILD_EXAMPLES)
  add_executable(flashvm_example_embedding examples/embedding/main.c)
  target_link_libraries(flashvm_example_embedding flashvm)
  add_executable(flashvm_example_extending examples/extending/main.c)
  target_link_libraries(flashvm_example_extending flashvm)
  if(FVM_BUILD_TESTS)
    add_test(NAME flashvm_example_embedding COMMAND flashvm_example_embedding)
    add_test(NAME flashvm_example_extending COMMAND flashvm_example_extending)
  endif()
endif()
if(FVM_BUILD_TESTS)
  add_executable(flashvm_interpreter tests/interpreter.c)
  target_link_libraries(flashvm_interpreter flashvm)
  add_test(NAME flashvm_interpreter COMMAND flashvm_interpreter)
endif()
if(FVM_BUILD_TESTS)
  add_executable(flashvm_swf_execute tests/swf_execute.c)
  target_link_libraries(flashvm_swf_execute flashvm)
  add_test(NAME flashvm_swf_execute COMMAND flashvm_swf_execute)
endif()
