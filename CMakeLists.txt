cmake_minimum_required(VERSION 2.8.12)
project(FlashVM C CXX)
option(FVM_BUILD_TESTS "Build tests" ON)
option(FVM_ENABLE_JIT "Build JIT components when available" OFF)
option(FVM_ENABLE_RENDER "Build rendering components when available" OFF)
option(FVM_ENABLE_MEDIA "Build media components when available" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++98 -Wall -Wextra -pedantic")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -std=c89 -Wall -Wextra -pedantic")
endif()
include_directories(${PROJECT_SOURCE_DIR}/include)
add_library(flashvm STATIC src/core/runtime.cpp src/capi/flashvm_c.cpp src/capi/value_c.cpp src/swf/parser.cpp src/swf/swf_c.cpp src/avm1/decoder.cpp src/avm1/interpreter.cpp src/capi/input_c.cpp)
if(FVM_BUILD_TESTS)
  enable_testing()
  add_executable(flashvm_smoke tests/smoke.c)
  target_link_libraries(flashvm_smoke flashvm)
  add_test(NAME flashvm_smoke COMMAND flashvm_smoke)
endif()
if(FVM_BUILD_TESTS)
  add_executable(flashvm_objects tests/objects.c)
  target_link_libraries(flashvm_objects flashvm)
  add_test(NAME flashvm_objects COMMAND flashvm_objects)
endif()

if(FVM_BUILD_TESTS)
 add_executable(flashvm_swf tests/swf.c)
 target_link_libraries(flashvm_swf flashvm)
 add_test(NAME flashvm_swf COMMAND flashvm_swf)
endif()

if(FVM_BUILD_TESTS)
 add_executable(flashvm_avm1 tests/avm1.c)
 target_link_libraries(flashvm_avm1 flashvm)
 add_test(NAME flashvm_avm1 COMMAND flashvm_avm1)
endif()

  endif()
endif()
if(FVM_BUILD_TESTS)

if(FVM_BUILD_TESTS)

  add_test(NAME flashvm_swf_execute COMMAND flashvm_swf_execute)
endif()

option(FVM_ENABLE_SQLITE "Enable optional SQLite adapter" ON)
if(FVM_ENABLE_SQLITE)
  find_package(SQLite3 QUIET)
  if(SQLite3_FOUND OR SQLITE3_FOUND)
    target_sources(flashvm PRIVATE src/capi/sqlite_c.cpp)
    if(SQLite3_INCLUDE_DIRS)
      target_include_directories(flashvm PRIVATE ${SQLite3_INCLUDE_DIRS})
    endif()
    if(SQLite3_LIBRARIES)
      target_link_libraries(flashvm ${SQLite3_LIBRARIES})
    else()
      target_link_libraries(flashvm sqlite3)
    endif()
    add_definitions(-DFVM_HAS_SQLITE=1)
    if(FVM_BUILD_TESTS)
      add_executable(flashvm_sqlite_test tests/sqlite.c)
      target_link_libraries(flashvm_sqlite_test flashvm)
      add_test(NAME flashvm_sqlite_test COMMAND flashvm_sqlite_test)
    endif()
  else()
    message(STATUS "SQLite not found; adapter disabled")
  endif()
endif()
if(FVM_BUILD_TESTS)
  add_executable(flashvm_input_test tests/input.c)

  add_test(NAME flashvm_input_test COMMAND flashvm_input_test)
