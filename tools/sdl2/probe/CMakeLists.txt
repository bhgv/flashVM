cmake_minimum_required(VERSION 3.10)
project(flashvm_sdl_sdk_probe C)
find_library(FVM_X11_LIBRARY NAMES X11 libX11.so.6)
find_library(FVM_XEXT_LIBRARY NAMES Xext libXext.so.6)
find_library(FVM_ALSA_LIBRARY NAMES asound libasound.so.2)
file(WRITE "${CMAKE_BINARY_DIR}/native-libraries.txt"
  "${FVM_X11_LIBRARY}\n${FVM_XEXT_LIBRARY}\n${FVM_ALSA_LIBRARY}\n")
find_package(SDL2 2.0.9 CONFIG QUIET)
if(SDL2_FOUND AND TARGET SDL2::SDL2)
  file(WRITE "${CMAKE_BINARY_DIR}/sdk-directory.txt" "${SDL2_DIR}\n")
  add_executable(sdk-probe main.c)
  target_link_libraries(sdk-probe PRIVATE SDL2::SDL2)
  set_target_properties(sdk-probe PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
else()
  file(WRITE "${CMAKE_BINARY_DIR}/sdk-directory.txt" "MISSING\n")
endif()
